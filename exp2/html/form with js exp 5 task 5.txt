    function calculateAndValidateIrrigation(event) {
      event.preventDefault(); // Prevent standard page refresh

      // Clear previous error/report outputs
      document.getElementById("error-container").innerText = "";
      document.getElementById("report-output").innerHTML = "";

      // Retrieve form values
      var farmerName = document.getElementById("name").value.trim();
      var mobile = document.getElementById("mobile").value.trim();
      var email = document.getElementById("email").value.trim();
      var plotID = document.getElementById("plotid").value.trim();
      var village = document.getElementById("village").value.trim();
      var cropStage = document.getElementById("cropstage").value.trim();
      var soilType = document.getElementById("soiltype").value.trim();
      var method = document.getElementById("method").value.trim();
      var currentSoilMoisture = parseFloat(document.getElementById("moisture").value);
      var requiredSoilMoisture = parseFloat(document.getElementById("req_moisture").value);
      var irrigationDate = document.getElementById("date").value;
      var photo = document.getElementById("photo").value;

      // -------------------------------------------------------------
      // TASK 5 & TASK 6: VALIDATION RULES
      // -------------------------------------------------------------

      // 1. Task 6: Ensure no required field is empty
      if (!farmerName || !mobile || !email || !plotID || !village || 
          !cropStage || !soilType || !method || isNaN(currentSoilMoisture) || 
          isNaN(requiredSoilMoisture) || !irrigationDate || !photo) {
        showError("Validation Error: Please fill in all required fields.");
        return false;
      }

      // 2. Task 5: Farmer Name - Only alphabets and spaces
      var nameRegex = /^[A-Za-z\s]+$/;
      if (!nameRegex.test(farmerName)) {
        showError("Validation Error: Farmer Name can only contain alphabets and spaces.");
        return false;
      }

      // 3. Task 5 & 6: Mobile Number - Exactly 10 digits
      var mobileRegex = /^\d{10}$/;
      if (!mobileRegex.test(mobile)) {
        showError("Validation Error: Mobile Number must be exactly 10 digits.");
        return false;
      }

      // 4. Task 5 & 6: Plot ID - Format: AGR-1234
      var plotRegex = /^AGR-\d{4}$/;
      if (!plotRegex.test(plotID)) {
        showError("Validation Error: Plot ID must follow the format 'AGR-1234' (e.g., AGR-5678).");
        return false;
      }

      // 5. Task 6: Soil Moisture - Must be between 0 and 100
      if (currentSoilMoisture < 0 || currentSoilMoisture > 100) {
        showError("Validation Error: Current Soil Moisture must be between 0 and 100.");
        return false;
      }

      if (requiredSoilMoisture < 0 || requiredSoilMoisture > 100) {
        showError("Validation Error: Required Soil Moisture must be between 0 and 100.");
        return false;
      }

      // -------------------------------------------------------------
      // CALCULATE IRRIGATION DEFICIT & DISPLAY RESULTS
      // -------------------------------------------------------------
      var moistureDeficit = requiredSoilMoisture - currentSoilMoisture;
      var statusMessage = "";

      if (moistureDeficit > 0) {
        statusMessage = "Irrigation Needed! Water Deficit: " + moistureDeficit.toFixed(1) + "%";
      } else {
        statusMessage = "Sufficient Soil Moisture. No Irrigation Required. Surplus/Balance: " + Math.abs(moistureDeficit).toFixed(1) + "%";
      }

      // Alert pop-up
      alert("Validation Successful!\n\nFarmer: " + farmerName + "\nPlot ID: " + plotID + "\nStatus: " + statusMessage);

      // Render Irrigation Status Report dynamically on the page
      var reportHTML = `
        <div style="background-color: #ffffff; border: 2px solid #5C8038; width: 60%; margin: 20px auto; padding: 20px; border-radius: 8px;">
          <h2 style='text-align:center; font-family:Arial; color:#5C8038;'>Irrigation Status Report</h2>
          <hr>
          <div style='font-size:16px; text-align:left; line-height: 1.6;'>
            <p><b>Farmer Name:</b> ${farmerName}</p>
            <p><b>Mobile:</b> ${mobile}</p>
            <p><b>Plot ID:</b> ${plotID}</p>
            <p><b>Crop Stage:</b> ${cropStage}</p>
            <p><b>Irrigation Date:</b> ${irrigationDate}</p>
            <p><b>Current Soil Moisture:</b> ${currentSoilMoisture}%</p>
            <p><b>Required Soil Moisture:</b> ${requiredSoilMoisture}%</p>
            <p><b>Moisture Deficit:</b> ${moistureDeficit.toFixed(1)}%</p>
            <p><b>Status:</b> ${statusMessage}</p>
          </div>
        </div>
      `;

      document.getElementById("report-output").innerHTML = reportHTML;
      return true;
    }

    function showError(msg) {
      document.getElementById("error-container").innerText = msg;
    }
  </script>
