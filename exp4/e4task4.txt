<script>
function validateForm() {
  let isValid = true;

  
  clearErrors();

  
  function showError(inputEl, errorElId, message) {
    document.getElementById(errorElId).innerText = message;
    inputEl.classList.add("invalid-field");
    isValid = false;
  }

  
  const nameInput = document.getElementById("name");
  const mobileInput = document.getElementById("mobile");
  const emailInput = document.getElementById("email");
  const plotidInput = document.getElementById("plotid");
  const villageInput = document.getElementById("village");
  const cropstageInput = document.getElementById("cropstage");
  const soiltypeInput = document.getElementById("soiltype");
  const methodInput = document.getElementById("method");
  const dateInput = document.getElementById("date");

  
  const nameVal = nameInput.value.trim();
  const mobileVal = mobileInput.value.trim();
  const emailVal = emailInput.value.trim();
  const plotidVal = plotidInput.value.trim();
  const villageVal = villageInput.value.trim();

  nameInput.value = nameVal;
  mobileInput.value = mobileVal;
  emailInput.value = emailVal;
  plotidInput.value = plotidVal;
  villageInput.value = villageVal;

  
  const nameRegex = /^[A-Za-z\s]+$/;
  if (nameVal === "") {
    showError(nameInput, "nameError", "Farmer Name is required.");
  } else if (!nameRegex.test(nameVal)) {
    showError(nameInput, "nameError", "Name must contain letters and spaces only.");
  }

  if (plotidVal === "") {
    showError(plotidInput, "plotidError", "Plot ID is required.");
  }

  if (villageVal === "") {
    showError(villageInput, "villageError", "Village is required.");
  }

  
  const mobileRegex = /^[6-9]\d{9}$/;
  if (mobileVal === "") {
    showError(mobileInput, "mobileError", "Mobile Number is required.");
  } else if (!mobileRegex.test(mobileVal)) {
    showError(mobileInput, "mobileError", "Enter a valid 10-digit Indian mobile number.");
  }

  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (emailVal === "") {
    showError(emailInput, "emailError", "Email address is required.");
  } else if (!emailRegex.test(emailVal)) {
    showError(emailInput, "emailError", "Please enter a valid email address.");
  }



  if (cropstageInput.value === "") {
    showError(cropstageInput, "cropstageError", "Please select a Crop Stage.");
  }

  if (soiltypeInput.value === "") {
    showError(soiltypeInput, "soiltypeError", "Please select a Soil Type.");
  }

  if (methodInput.value === "") {
    showError(methodInput, "methodError", "Please select an Irrigation Method.");
  }

  
  if (dateInput.value === "") {
    showError(dateInput, "dateError", "Registration Date is required.");
  } else {
    <!-- Compare entered date against today's date -->
    const selectedDate = new Date(dateInput.value);
    const today = new Date();
    
    today.setHours(0, 0, 0, 0);

    if (selectedDate > today) {
      showError(dateInput, "dateError", "Registration Date cannot be in the future.");
    }
  }

  return isValid;
}

function clearErrors() {
  
  const errorSpans = document.querySelectorAll(".error-msg");
  errorSpans.forEach(span => span.innerText = "");

  
  const inputs = document.querySelectorAll("input, select");
  inputs.forEach(input => input.classList.remove("invalid-field"));
}
</script>
